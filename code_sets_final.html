Q1Implement the algorithm which finds the LOC of the given item in an array DATA with n > 1 element.
#include <stdio.h>

int main()
{
    int a[100];      // array
    int n, x;        // n = total element, x = search item
    int i, loc = -1; // i = loop, loc = location

    printf("Enter number of elements: ");
    scanf("%d", &n);

    printf("Enter array elements:\n");

    for(i = 0; i < n; i++)
    {
        scanf("%d", &a[i]);
    }

    printf("Enter item to search: ");
    scanf("%d", &x);

    for(i = 0; i < n; i++)
    {
        if(a[i] == x)
        {
            loc = i + 1;
            break;
        }
    }

    if(loc == -1)
        printf("Item not found\n");
    else
        printf("Location = %d\n", loc);

    return 0;
}
Q3Implement the algorithm which counts the number of occurrences of a given word in a given line of text.
#include <stdio.h>
#include <string.h>

int main()
{
    char t[200];   // t = text
    char w[50];    // w = word to find
    int cnt = 0;   // cnt = count
    char *p;       // p = pointer for searching

    printf("Enter a line of text:\n");
    gets(t);

    printf("Enter word to count: ");
    scanf("%s", w);

    p = t;

    while((p = strstr(p, w)) != NULL)
    {
        cnt++;
        p = p + strlen(w);
    }

    printf("Number of occurrences = %d\n", cnt);

    return 0;
}

Q1Implement the algorithm which inserts a given ITEM into a given location in the array.
#include <stdio.h>

int main()
{
    int a[100];   // array
    int n, x, p;  // n = elements, x = item, p = position
    int i;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    printf("Enter array elements:\n");

    for(i = 0; i < n; i++)
    {
        scanf("%d", &a[i]);
    }

    printf("Enter position: ");
    scanf("%d", &p);

    printf("Enter item: ");
    scanf("%d", &x);

    for(i = n; i >= p; i--)
    {
        a[i] = a[i - 1];
    }

    a[p - 1] = x;

    n++;

    printf("Array after insertion:\n");

    for(i = 0; i < n; i++)
    {
        printf("%d ", a[i]);
    }

    return 0;
}
Q2Implement the algorithm to delete a given item in the linked list. Implement the linked list using parallel arrays.
#include <stdio.h>

int main()
{
    int d[10] = {10, 20, 30, 40, 50};  // d = data
    int nx[10] = {1, 2, 3, 4, -1};     // nx = next index

    int st = 0;      // st = start
    int x;           // x = item to delete
    int i, pv = -1, c; // pv = previous, c = current

    printf("Enter item to delete: ");
    scanf("%d", &x);

    c = st;

    while(c != -1)
    {
        if(d[c] == x)
            break;

        pv = c;
        c = nx[c];
    }

    if(c == -1)
    {
        printf("Item not found\n");
    }
    else
    {
        if(pv == -1)
            st = nx[c];
        else
            nx[pv] = nx[c];

        printf("Item deleted successfully\n");
    }

    printf("Linked List:\n");

    c = st;

    while(c != -1)
    {
        printf("%d ", d[c]);
        c = nx[c];
    }

    return 0;
}

Q1Implement the binary search algorithm which finds the location LOC of the given ITEM in a given sorted array.
#include <stdio.h>

int main()
{
    int a[100];     // array
    int n, x;       // n = elements, x = item to search
    int low, high, mid;
    int f = 0;      // f = found
    int i;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    printf("Enter sorted array:\n");

    for(i = 0; i < n; i++)
    {
        scanf("%d", &a[i]);
    }

    printf("Enter item to search: ");
    scanf("%d", &x);

    low = 0;
    high = n - 1;

    while(low <= high)
    {
        mid = (low + high) / 2;

        if(a[mid] == x)
        {
            printf("Item found at LOC = %d\n", mid + 1);
            f = 1;
            break;
        }
        else if(x < a[mid])
        {
            high = mid - 1;
        }
        else
        {
            low = mid + 1;
        }
    }

    if(f == 0)
        printf("Item not found\n");

    return 0;
}
Q2Implement the algorithm to observe the insertion and deletion operation in a queue.
#include <stdio.h>

int main()
{
    int q[5];     // q = queue
    int f = 0;    // f = front
    int r = -1;   // r = rear
    int x;        // x = item
    int ch;       // ch = choice
    int i;

    while(1)
    {
        printf("\n1. Insert\n");
        printf("2. Delete\n");
        printf("3. Display\n");
        printf("4. Exit\n");

        printf("Enter choice: ");
        scanf("%d", &ch);

        if(ch == 1)
        {
            if(r == 4)
            {
                printf("Queue is full\n");
            }
            else
            {
                printf("Enter item: ");
                scanf("%d", &x);

                r++;
                q[r] = x;

                printf("Item inserted\n");
            }
        }

        else if(ch == 2)
        {
            if(f > r)
            {
                printf("Queue is empty\n");
            }
            else
            {
                printf("Deleted item = %d\n", q[f]);
                f++;
            }
        }

        else if(ch == 3)
        {
            if(f > r)
            {
                printf("Queue is empty\n");
            }
            else
            {
                for(i = f; i <= r; i++)
                {
                    printf("%d ", q[i]);
                }
                printf("\n");
            }
        }

        else if(ch == 4)
        {
            break;
        }

        else
        {
            printf("Wrong choice\n");
        }
    }

    return 0;
}

Q1Implement the algorithm to delete a given item in the array.
#include <stdio.h>

int main()
{
    int a[100];   // array
    int n, x;     // n = elements, x = item to delete
    int i, p = -1; // p = position

    printf("Enter number of elements: ");
    scanf("%d", &n);

    printf("Enter array elements:\n");

    for(i = 0; i < n; i++)
    {
        scanf("%d", &a[i]);
    }

    printf("Enter item to delete: ");
    scanf("%d", &x);

    for(i = 0; i < n; i++)
    {
        if(a[i] == x)
        {
            p = i;
            break;
        }
    }

    if(p == -1)
    {
        printf("Item not found\n");
    }
    else
    {
        for(i = p; i < n - 1; i++)
        {
            a[i] = a[i + 1];
        }

        n--;

        printf("Array after deletion:\n");

        for(i = 0; i < n; i++)
        {
            printf("%d ", a[i]);
        }
    }

    return 0;
}
Q3Implement the algorithm to observe the PUSH and POP operation in a stack.
#include <stdio.h>

int main()
{
    int s[5];     // s = stack
    int top = -1;
    int x;        // x = item
    int ch;       // ch = choice

    while(1)
    {
        printf("\n1. PUSH\n");
        printf("2. POP\n");
        printf("3. Exit\n");

        printf("Enter choice: ");
        scanf("%d", &ch);

        if(ch == 1)
        {
            if(top == 4)
            {
                printf("Stack is full\n");
            }
            else
            {
                printf("Enter item: ");
                scanf("%d", &x);

                top++;
                s[top] = x;

                printf("Item pushed\n");
            }
        }

        else if(ch == 2)
        {
            if(top == -1)
            {
                printf("Stack is empty\n");
            }
            else
            {
                printf("Popped item = %d\n", s[top]);
                top--;
            }
        }

        else if(ch == 3)
        {
            break;
        }

        else
        {
            printf("Wrong choice\n");
        }
    }

    return 0;
}

Q1Implement the algorithm to insert a new item in the linked list. Demonstrate the linked list using parallel arrays.
#include <stdio.h>

int main()
{
    int d[10] = {10, 20, 30};  // d = data
    int nx[10] = {1, 2, -1};   // nx = next index

    int st = 0;   // st = start
    int x;        // x = new item
    int N = 3;    // N = new node position
    int c;        // c = current

    printf("Enter new item: ");
    scanf("%d", &x);

    d[N] = x;
    nx[N] = -1;

    c = st;

    while(nx[c] != -1)
    {
        c = nx[c];
    }

    nx[c] = N;

    printf("Linked List after insertion:\n");

    c = st;

    while(c != -1)
    {
        printf("%d ", d[c]);
        c = nx[c];
    }

    return 0;
}
Q3A graph G is represented by means of number of M nodes such that 1,2,...M and a list of N ordered pairs of integers, representing the edges of G. Write a program to find the M x M adjacency matrix of the graph G.
#include <stdio.h>

int main()
{
    int mat[10][10]; // mat = adjacency matrix
    int m, e;        // m = nodes, e = edges
    int a, b;        // a, b = edge pair
    int i, j;

    printf("Enter number of nodes (M): ");
    scanf("%d", &m);

    printf("Enter number of edges (N): ");
    scanf("%d", &e);

    for(i = 0; i < m; i++)
    {
        for(j = 0; j < m; j++)
        {
            mat[i][j] = 0;
        }
    }

    printf("Enter edges (as pairs, e.g. 1 2):\n");

    for(i = 0; i < e; i++)
    {
        scanf("%d %d", &a, &b);

        mat[a - 1][b - 1] = 1;
        mat[b - 1][a - 1] = 1;
    }

    printf("Adjacency Matrix:\n");

    for(i = 0; i < m; i++)
    {
        for(j = 0; j < m; j++)
        {
            printf("%d ", mat[i][j]);
        }

        printf("\n");
    }

    return 0;
}
